const url = "https://docs.google.com/spreadsheets/d/e/2PACX-1vTa1OMKOtrrTqrSXDtDkHy-23TEcdkMPRUObduI0_8wasPPvO4XC5n1u8oOpXf3qkyq5U9qeIoeHKaF/pub?gid=0&single=true&output=csv"; // tu URL de Google Sheets fetch(url) .then(response => response.text()) .then(csv => { // Separador correcto: punto y coma const rows = csv.split("\n").map(r => r.split(";")); // Cabeceras reales del CSV const headers = rows[0]; // Crear cabecera de la tabla automáticamente const thead = document.querySelector("thead"); const headerRow = document.createElement("tr"); headers.forEach(h => { const th = document.createElement("th"); th.textContent = h; headerRow.appendChild(th); }); thead.appendChild(headerRow); // Crear filas con todos los datos const tbody = document.querySelector("tbody"); rows.slice(1).forEach(row => { const tr = document.createElement("tr"); row.forEach(cell => { const td = document.createElement("td"); td.textContent = cell; tr.appendChild(td); }); tbody.appendChild(tr); }); // FILTRO por CodEquipo const filtro = document.getElementById("filtro-equipo"); filtro.addEventListener("change", () => { const valor = filtro.value; tbody.innerHTML = ""; rows.slice(1).forEach(row => { const equipo = row[headers.indexOf("CodEquipo")]; if (valor === "" || equipo === valor) { const tr = document.createElement("tr"); row.forEach(cell => { const td = document.createElement("td"); td.textContent = cell; tr.appendChild(td); }); tbody.appendChild(tr); } }); }); });
